# Types

Types exported by `run`.

```ts
import type { RunInput, RunnerOptions } from 'run';

const input: RunInput = {
  source: 'return 6 * 7;',
};

const options: RunnerOptions = {
  limits: { timeoutMs: 10_000 },
};
```
