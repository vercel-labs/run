# `getBindingContext()`

Get the context for the currently executing host binding.

```ts
import { getBindingContext, run } from 'run';

await run({
  source: 'return await users.find("user_1");',
  bindings: {
    users: {
      find: async (id: string) => {
        const { abortSignal } = getBindingContext();
        return database.users.find(id, { signal: abortSignal });
      },
    },
  },
});
```

## Import

```ts
import { getBindingContext } from 'run';
```
